/opt/bitnami/kafka/bin/kafka-topics.sh --create --topic messages --bootstrap-server localhost:9092 --partitions 3 --replication-factor 2 

/opt/bitnami/kafka/bin/kafka-topics.sh --create --topic filtered_messages --bootstrap-server localhost:9092 --partitions 3 --replication-factor 2
